```swift
extension/inside enum HexFlowerLayout {
    /// rotation: radians, positive = clockwise. Default 0 keeps today's positions.
    static func outerOffsets(radius: CGFloat = GameTheme.outerRingRadius, rotation: Double = 0) -> [CGSize]
    /// Extra hit area around the flower square for grabbing the ring (D-01). Must stay below GameTheme.xl (32).
    static let spinZoneMargin: CGFloat = 28
    /// Side of the padded square the grid's gesture lives in: flowerDiameter() + 2 * spinZoneMargin (= 350).
    static func spinZoneSide() -> CGFloat
    /// Center of the flower in the padded gesture view's LOCAL coordinates: (side/2, side/2).
    static var spinZoneCenter: CGPoint { get }
    /// -1 = center tile, 0...5 = outer tile by CURRENT index (outerLetters order), nil = off every tile (spin zone).
    static func tileIndex(at point: CGPoint, center: CGPoint, rotation: Double = 0,
                          hexSize: CGFloat = GameTheme.hexSize, radius: CGFloat = GameTheme.outerRingRadius) -> Int?
    /// atan2(point.y - center.y, point.x - center.x): radians, clockwise-positive.
    static func touchAngle(of point: CGPoint, center: CGPoint) -> Double
    /// True within hexSize / 2 of the center, where atan2 is unstable (ignore spin samples there).
    static func isInCenterDeadZone(_ point: CGPoint, center: CGPoint, hexSize: CGFloat = GameTheme.hexSize) -> Bool
    /// Cyclic rotation by k slots clockwise: new[(i + k) mod n] = old[i]. Negative k = counter-clockwise.
    static func rotated<T>(_ items: [T], bySlots k: Int) -> [T]
}
```
