// AngularVelocityTracker.swift
struct AngularVelocityTracker {
    var window: TimeInterval        // default 0.1  : only samples this recent count toward release velocity
    var minSpan: TimeInterval       // default 0.016: need at least this much time between first and last sample
    var maxSpeed: Double            // default 24   : clamp (rad/s)
    var restSpeed: Double           // default 0.3  : |w| below this reports 0
    init(window: TimeInterval = 0.1, minSpan: TimeInterval = 0.016, maxSpeed: Double = 24, restSpeed: Double = 0.3)
    private(set) var accumulatedAngle: Double   // sum of unwrapped deltas since reset; 0 at the first sample
    var sampleCount: Int { get }
    mutating func reset()
    mutating func add(rawAngle: Double, at time: Date)   // rawAngle = atan2 result in (-pi, pi]
    func velocity(at releaseTime: Date) -> Double         // signed rad/s
    static func normalizedDelta(_ delta: Double) -> Double // wraps into (-pi, pi]
}

// TickThrottle.swift
struct TickThrottle {
    struct Decision: Equatable { var haptic: Bool; var sound: Bool; static let none = Decision(haptic: false, sound: false) }
    var minHapticInterval: TimeInterval  // default 0.045
    var minSoundInterval: TimeInterval   // default 0.055
    var decimateAboveRate: Double        // default 18 (crossings per second)
    init(minHapticInterval: TimeInterval = 0.045, minSoundInterval: TimeInterval = 0.055, decimateAboveRate: Double = 18)
    mutating func register(crossings: Int, at time: TimeInterval, rate: Double) -> Decision
    mutating func reset()
}

// SpinSession.swift
struct SpinSession {
    static let jackpotRevolutions: Double = 5
    private(set) var isOpen: Bool
    private(set) var startAngle: Double
    init()
    mutating func begin(at angle: Double)            // no-op if already open (a re-grab mid-coast stays one session)
    mutating func end(finalAngle: Double) -> Bool    // closes; returns isJackpot(netAngle: finalAngle - startAngle); false if not open
    mutating func reset()
    static func isJackpot(netAngle: Double) -> Bool
}
