```swift
// RingSpinTuning.swift
struct RingSpinTuning: Equatable {
    var viscousFriction: Double = 0.45      // k (1/s): main duration knob
    var dryFriction: Double = 0.35          // c (rad/s^2): prize-wheel tail
    var handoffSpeed: Double = 1.5          // w_h (rad/s): coast hands off to the settle spring
    var settleFrequency: Double = 10        // L (1/s): critically damped settle
    var settleTimeConstants: Double = 8     // settle lasts settleTimeConstants / settleFrequency seconds
    var maxReleaseSpeed: Double = 24        // w_max (rad/s): release clamp == max-duration cap
    var restSpeed: Double = 0.3             // below this a release counts as "no flick"
    static let standard = RingSpinTuning()
    var settleDuration: TimeInterval { settleTimeConstants / settleFrequency }   // 0.8 s
}

// SlotMath.swift
enum SlotMath {
    static let slotAngle: Double = .pi / 3
    static func detent(_ angle: Double) -> Int                 // Int((angle / slotAngle).rounded())
    static func crossings(from a: Double, to b: Double) -> Int // abs(detent(b) - detent(a))
    static func slotsMoved(finalAngle: Double) -> Int          // detent(finalAngle), signed
    static func normalizedSlots(_ k: Int) -> Int               // ((k % 6) + 6) % 6
    static func nearestSlotAngle(_ angle: Double) -> Double    // Double(detent(angle)) * slotAngle
    static func settleTarget(angle: Double, velocity: Double, tuning: RingSpinTuning) -> Double
}

// RingSpinTrajectory.swift
struct RingSpinTrajectory: Equatable {
    let tuning: RingSpinTuning
    let startAngle: Double
    let releaseVelocity: Double      // after clamping to +-maxReleaseSpeed and zeroing |w| < restSpeed
    let handoffTime: TimeInterval
    let handoffAngle: Double
    let handoffVelocity: Double
    let targetAngle: Double          // exact multiple of SlotMath.slotAngle
    var duration: TimeInterval { handoffTime + tuning.settleDuration }
    init(startAngle: Double, velocity: Double, tuning: RingSpinTuning = .standard)
    func angle(at t: TimeInterval) -> Double      // t <= 0 -> startAngle; t >= duration -> targetAngle EXACTLY
    func velocity(at t: TimeInterval) -> Double   // t >= duration -> 0
    func isFinished(at t: TimeInterval) -> Bool   // t >= duration
}
```
