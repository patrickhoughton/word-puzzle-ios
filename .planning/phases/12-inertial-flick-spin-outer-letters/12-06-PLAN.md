enum RingSpinEvent: Equatable {
    case began                               // ring left rest (prepare haptics)
    case ticks(count: Int, speed: Double)    // detent crossings in this update; speed = |rad/s|
    case clunk                               // landed on a slot
    case settled(slots: Int)                 // signed slots this session (0 allowed); exactly once per session
    case jackpot                             // D-12, after .settled, cosmetic only
}
enum RingSpinPhase: Equatable { case idle, tracking, coasting }

@MainActor @Observable
final class RingSpinController {
    static let slop: CGFloat = 10                     // == EmptyDoubleTapDetector.maxTapTravel default
    private(set) var angle: Double                    // radians, clockwise; 0 at rest
    private(set) var phase: RingSpinPhase
    var isActive: Bool { phase != .idle }             // locks tiles (D-14)
    @ObservationIgnored var reduceMotion: Bool        // D-18
    @ObservationIgnored var onEvent: ((RingSpinEvent) -> Void)?
    @ObservationIgnored let tuning: RingSpinTuning
    init(tuning: RingSpinTuning = .standard,
         clock: @escaping () -> TimeInterval = { CACurrentMediaTime() },
         usesDisplayLink: Bool = true)
    func touchDown()                                                       // first onChanged of a spin-mode touch
    func track(point: CGPoint, start: CGPoint, center: CGPoint, time: Date)
    func release(time: Date)
    func cancelAndCommit()                                                 // D-15 / sheets / background
    func reset()                                                           // new round / leaving .playing: angle 0, no events
    func step()                                                            // display-link tick; tests call directly
}

@MainActor final class SpinFeedback {
    init(sound: SoundManager = .shared, clock: @escaping () -> TimeInterval = { CACurrentMediaTime() },
         throttle: TickThrottle = TickThrottle())
    func prepare()
    func ticks(count: Int, speed: Double, soundEnabled: Bool)
    func clunk(soundEnabled: Bool)
    func jackpot(soundEnabled: Bool)
}

    Create RingSpinController.swift (`import SwiftUI` and `import QuartzCore`) implementing the `<interfaces>` contract.

    Private state (all `@ObservationIgnored` except `angle` and `phase`): `clock`, `usesDisplayLink`, `tracker = AngularVelocityTracker(maxSpeed: tuning.maxReleaseSpeed, restSpeed: tuning.restSpeed)`, `session = SpinSession()`, `trajectory: RingSpinTrajectory?`, `releaseTime: TimeInterval = 0`, `trackStartAngle: Double = 0`, `didExceedSlop = false`, `ignoresCurrentTouch = false`, `displayLink: CADisplayLink?`.

    Methods (exact semantics):
    - `touchDown()`: `ignoresCurrentTouch = false; didExceedSlop = false`; if `phase == .coasting` { stopDisplayLink(); trajectory = nil; phase = .tracking } (grab to stop, D-13: angle stays where it is).
    - `track(point:start:center:time:)`:
      ```
      guard !ignoresCurrentTouch else { return }
      if !didExceedSlop {
          guard hypot(point.x - start.x, point.y - start.y) >= Self.slop else { return }
          didExceedSlop = true
          if phase == .idle { session.begin(at: angle); phase = .tracking; onEvent?(.began) }
          trackStartAngle = angle
          tracker.reset()
      }
      guard !HexFlowerLayout.isInCenterDeadZone(point, center: center) else { return }
      tracker.add(rawAngle: HexFlowerLayout.touchAngle(of: point, center: center), at: time)
      let newAngle = trackStartAngle + tracker.accumulatedAngle
      emitTicks(from: angle, to: newAngle, speed: abs(tracker.velocity(at: time)))
      angle = newAngle
      ```
      (Reference set at the slop-crossing point: the first sample contributes 0, so there is no visible jump, Pitfall 3.)
    - `release(time:)`:
      ```
      defer { didExceedSlop = false }
      guard !ignoresCurrentTouch, phase == .tracking else { return }   // idle tap or already coasting: no-op (idempotent)
      if reduceMotion { finish(at: SlotMath.nearestSlotAngle(angle), allowJackpot: true); return }   // D-18
      let w = didExceedSlop ? tracker.velocity(at: time) : 0          // a grab released without travel settles in place
      trajectory = RingSpinTrajectory(startAngle: angle, velocity: w, tuning: tuning)
      releaseTime = clock()
      phase = .coasting
      startDisplayLink()
      ```
    - `step()`:
      ```
      guard phase == .coasting, let trajectory else { return }
      let t = clock() - releaseTime
      let newAngle = trajectory.angle(at: t)
      emitTicks(from: angle, to: newAngle, speed: abs(trajectory.velocity(at: t)))
      angle = newAngle
      if trajectory.isFinished(at: t) { finish(at: trajectory.targetAngle, allowJackpot: true) }
      ```
    - `cancelAndCommit()`: `guard phase != .idle else { return }; ignoresCurrentTouch = true; finish(at: SlotMath.nearestSlotAngle(angle), allowJackpot: false)`.
    - `reset()`: stopDisplayLink(); trajectory = nil; session.reset(); tracker.reset(); didExceedSlop = false; ignoresCurrentTouch = true; phase = .idle; angle = 0. No events.
    - `private func emitTicks(from:to:speed:)`: `let c = SlotMath.crossings(from: a, to: b); if c > 0 { onEvent?(.ticks(count: c, speed: speed)) }`.
    - `private func finish(at target: Double, allowJackpot: Bool)`:
      ```
      stopDisplayLink(); trajectory = nil
      angle = target                                       // exact multiple of pi/3 (Pitfall 10)
      onEvent?(.clunk)
      let slots = SlotMath.slotsMoved(finalAngle: target)  // session always starts at angle 0 (reset after every commit)
      let isJackpot = session.end(finalAngle: target) && allowJackpot
      tracker.reset()
      phase = .idle
      // Pitfall 1: commit + angle reset in ONE animation-disabled pass. With letter-keyed tiles the
      // on-screen positions are identical before and after, so the commit is a visual no-op.
      var transaction = Transaction()
      transaction.disablesAnimations = true
      withTransaction(transaction) {
          onEvent?(.settled(slots: slots))
          angle = 0
      }
      if isJackpot { onEvent?(.jackpot) }
      ```
