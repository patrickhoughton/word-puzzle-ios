```swift
    /// D-07: longest run of consecutive LOCAL calendar days with >= 1 finished round, over FULL
    /// history (no 400-day window; the data is small). Same day logic as currentStreak():
    /// injected calendar + startOfDay, adjacency via date(byAdding: .day) so DST days count as one day.
    func longestStreak() -> Int {
        let records = (try? context.fetch(FetchDescriptor<GameRecord>())) ?? []
        let days = Set(records.map { calendar.startOfDay(for: $0.date) }).sorted()
        var best = 0, run = 0
        var previous: Date?
        for day in days {
            if let previous, calendar.date(byAdding: .day, value: 1, to: previous) == day { run += 1 } else { run = 1 }
            best = max(best, run)
            previous = day
        }
        return best
    }

    /// D-09/D-23: whole number, rounded to nearest; nil when no finished games (never divide by zero, D-20).
    func averageScore() -> Int? {
        let records = (try? context.fetch(FetchDescriptor<GameRecord>())) ?? []
        guard !records.isEmpty else { return nil }
        return Int((Double(records.reduce(0) { $0 + $1.score }) / Double(records.count)).rounded())
    }

    func averageWordsPerGame() -> Int? { /* same shape over wordsFoundCount */ }

    /// D-14: highest stored tier; pre-Phase-9 rows (nil) are ignored (D-12). Unknown raw values map to nil.
    func bestRank() -> RankTier? {
        let records = (try? context.fetch(FetchDescriptor<GameRecord>())) ?? []
        return records.compactMap(\.rankRaw).max().flatMap(RankTier.init(rawValue:))
    }

    /// D-10/D-13: finished rounds only; nil (pre-Phase-9) contributes 0.
    func totalPangramsFound() -> Int { /* reduce { $0 + ($1.pangramsFound ?? 0) } */ }

    /// D-10: number of finished rounds whose pangram sweep fired.
    func totalSweeps() -> Int { /* records.filter { $0.hadSweep == true }.count */ }

    /// D-21: true when a round was FINISHED today (GameRecord, not RoundStartRecord — streaks derive
    /// from GameRecord). Uses the same todayBounds() boundary as every other "today" query.
    func hasFinishedRoundToday(now: Date = .now) -> Bool {
        let (startOfDay, startOfNextDay) = todayBounds(now: now)
        let descriptor = FetchDescriptor<GameRecord>(
            predicate: #Predicate { $0.date >= startOfDay && $0.date < startOfNextDay }
        )
        return ((try? context.fetchCount(descriptor)) ?? 0) > 0
    }
```
