```swift
import Foundation
import Observation

enum TutorialAction: Equatable {
    case letter(Character), delete, clearWord, submit, shuffle, scoreBar, finish, topBar
}
enum TutorialTarget: Equatable {
    case letter(Character), wordDisplay, delete, shuffle, scoreBar, finish
}
enum TutorialMode: Equatable { case firstLaunch, replay }

/// Delays are injected so unit tests run with zero waits. A `.zero` delay applies synchronously.
struct TutorialTiming {
    /// Build step: let the finished word sit on screen before step 2 clears it.
    var stepAdvanceDelay: Duration = .milliseconds(600)
    /// D-06 / Pitfall 5: shake (~0.4s) + "Forgot the middle!" must be seen BEFORE the rule is explained.
    var missExplainDelay: Duration = .milliseconds(900)
    /// Let the Phase 8 length pill + sweep start before the banner moves on.
    var pangramAdvanceDelay: Duration = .milliseconds(1500)
    /// Pitfall 8: idle time on the pangram step before each hint level.
    var pangramHintDelay: Duration = .seconds(20)
    static let standard = TutorialTiming()
    static let immediate = TutorialTiming(stepAdvanceDelay: .zero, missExplainDelay: .zero,
                                          pangramAdvanceDelay: .zero, pangramHintDelay: .seconds(3600))
}

@MainActor
@Observable
final class TutorialController {
    /// D-07: a SEPARATE view-model with NO persistence store. Every write in GameViewModel is
    /// `persistenceStore?.…`, so this board cannot record a round start or a GameRecord, and
    /// `requestNextRound` (the paywall gate) is never called on it.
    let practice: GameViewModel
    private(set) var isActive = false
    private(set) var mode: TutorialMode = .firstLaunch
    private(set) var step: TutorialStep = .build
    /// Step 2 sub-state: the guided miss has happened and the rule is now explained.
    private(set) var hasMissed = false
    /// Step 4 sub-state: Delete done, now tap the word to clear.
    private(set) var hasDeleted = false
    /// Step 7 hints: 0 none, 1 text hint, 2 follow-the-glow.
    private(set) var pangramHintLevel = 0
    /// True while a delayed transition is pending; all guided input is ignored meanwhile.
    private(set) var isWaiting = false
    /// Set by WordPuzzleApp. Called after the flag is set and isActive is false.
    @ObservationIgnored var onExit: ((TutorialMode) -> Void)?

    init(wordList: WordList, defaults: UserDefaults = .standard, timing: TutorialTiming = .standard)

    var isGuided: Bool { isActive && step != .ready }
    var copy: TutorialCopy { TutorialText.copy(for: step, hasMissed: hasMissed, pangramHintLevel: pangramHintLevel) }
    var highlightTarget: TutorialTarget? { ... }
    /// Dim non-target controls/tiles: guided steps, except the free-form pangram step until hint level 2.
    var dimsBoard: Bool { guard isGuided else { return false }; return step == .pangram ? pangramHintLevel >= 2 : true }

    func begin(mode: TutorialMode)
    func allows(_ action: TutorialAction) -> Bool
    @discardableResult func send(_ action: TutorialAction) -> Bool
    func foundWordsDismissed()
    func skip()
}
```
Implementation rules:
- `init` stores `wordList`, `defaults`, `timing`; `practice = GameViewModel(wordList: wordList, persistenceStore: nil)`.
- `begin(mode:)`: cancel pending/hint tasks; `self.mode = mode`; if `!wordList.isLoaded` -> UI-SPEC error state: `TutorialFlag.markSeen(in: defaults); isActive = false; onExit?(mode); return`. Else `practice.startNewRound(with: PracticePuzzle.puzzle)`, reset `step = .build, hasMissed = false, hasDeleted = false, pangramHintLevel = 0, pangramRejections = 0, isWaiting = false`, then `isActive = true`. Never persist the step (D-13).
- Scripted target per step: build/swipe -> `PracticePuzzle.buildWord`; centerMiss (before miss) -> `missWord`; drag -> `dragWord`; pangram at hint level 2 -> `pangram`. `nextExpected`: if `practice.currentWord.lowercased()` is a prefix of the target and shorter, return the target's character at that index, else nil.
- `allows(_:)`: if `!isActive || step == .ready` return true. If `isWaiting` return false. Then per step:
  - build: `.letter(c)` iff `c == nextExpected`; everything else false.
  - centerMiss, !hasMissed: `.letter(c)` iff next expected of "hold"; `.submit` iff currentWord == "hold".
  - centerMiss, hasMissed: `.letter(c)` iff `c == PracticePuzzle.centerLetter` and currentWord is empty.
  - swipe / drag: `.letter` iff next expected; `.submit` iff currentWord == target.
  - fixMistake: `.delete` iff !hasDeleted; `.clearWord` iff hasDeleted.
  - shuffle: `.shuffle` only.
  - pangram: `.letter`, `.delete`, `.clearWord`, `.submit`, `.shuffle` true; `.scoreBar`, `.finish`, `.topBar` false.
  - scoreBar: `.scoreBar` only.
  Compare letters case-insensitively (lowercase both).
- `send(_:)`: `guard isActive, allows(action) else { return false }` then:
  - `.letter(c)`: `practice.append(c)`. build: if currentWord == buildWord -> `after(timing.stepAdvanceDelay) { practice.clearCurrentWord(); step = .centerMiss }`. centerMiss && hasMissed -> `step = .swipe` (keep the "p" in the word).
  - `.submit`: `practice.submitCurrentWord()`, then by step: centerMiss with `lastOutcome == .rejected(.missingCenterLetter)` -> `after(timing.missExplainDelay) { hasMissed = true }`; swipe with `.accepted` -> `step = .fixMistake`, then `for ch in PracticePuzzle.fixPrefill { practice.append(ch) }`; drag with `.accepted` -> `step = .shuffle`; pangram: if `case .accepted(_, _, true)` -> cancel hint task, `after(timing.pangramAdvanceDelay) { step = .scoreBar }`; if `.rejected` -> `pangramRejections += 1`, `pangramHintLevel = max(pangramHintLevel, pangramRejections >= 4 ? 2 : (pangramRejections >= 2 ? 1 : 0))`.
  - `.delete`: `practice.deleteLast()`; fixMistake -> `hasDeleted = true`.
  - `.clearWord`: `practice.clearCurrentWord()`; fixMistake && hasDeleted -> `step = .drag`.
  - `.shuffle`: `let did = practice.shuffleOuterLetters()`; if did && step == .shuffle -> `step = .pangram; startHintTimer()`. Return `did`.
  - `.scoreBar`, `.topBar`: return true (GameView presents the sheet; advancement comes from `foundWordsDismissed()`).
  - `.finish`: `exit()`; return true.
  Return true after a performed action.
- `foundWordsDismissed()`: if isActive && step == .scoreBar -> `step = .ready` (Pitfall 10: advance on DISMISS so the Ready card is not hidden behind the sheet).
- `after(_ delay: Duration, _ work: @escaping () -> Void)`: if `delay == .zero` run `work()` now. Else set `isWaiting = true`, cancel any previous pending task, and start `Task { @MainActor [weak self] in try? await Task.sleep(for: delay); guard let self, !Task.isCancelled, self.isActive else { return }; self.isWaiting = false; work() }`. Keep the handle in an `@ObservationIgnored private var pendingTask: Task<Void, Never>?`.
- `startHintTimer()`: `@ObservationIgnored private var hintTask`; sleeps `pangramHintDelay`, then if still on .pangram sets level to max(level, 1); sleeps again, sets max(level, 2).
- `highlightTarget`: nil if !isActive or isWaiting. build/centerMiss(!hasMissed)/swipe/drag: if currentWord == target -> `.wordDisplay` (build: nil, it auto-advances) else `.letter(next)`. centerMiss(hasMissed) -> `.letter(PracticePuzzle.centerLetter)`. fixMistake -> hasDeleted ? `.wordDisplay` : `.delete`. shuffle -> `.shuffle`. pangram -> level >= 2 ? (next expected of "dolphin" -> `.letter`, else if currentWord non-empty `.wordDisplay`) : nil. scoreBar -> `.scoreBar`. ready -> `.finish`.
- `skip()`: `exit()`. `private func exit()`: cancel tasks, `isWaiting = false`, `TutorialFlag.markSeen(in: defaults)` (D-10/D-11), `isActive = false`, `onExit?(mode)`.
- Never call `practice.finishRound()` or `practice.requestNextRound` (D-07: no missed-words cover, no paywall).
