struct Puzzle {
    let letters: Set<Character>   // exactly 7 unique letters
    let centerLetter: Character   // one of `letters`
    let validWords: [String]      // all words: >=4 chars, contain centerLetter, subset of letters
    let pangrams: [String]        // words whose Set == letters

}

enum GeneratorError: Error {
    case exhaustedRetries
    case emptyWordList
}

// WordList.swift
@Observable
final class WordList {
    private(set) var words: Set<String>
    private(set) var pangramWords: [String]   // cached: words with exactly 7 unique letters and >=7 chars
    private(set) var isLoaded: Bool
    func load() async            // reads enable-clean.txt from bundle on a background Task
    func contains(_ word: String) -> Bool   // O(1) Set lookup, lowercased
}

// PuzzleGenerator.swift
func generatePuzzle(from wordList: WordList, maxAttempts: Int = 1000) throws -> Puzzle

```swift
import Foundation

func generatePuzzle(from wordList: WordList, maxAttempts: Int = 1000) throws -> Puzzle {
    let pangramWords = wordList.pangramWords          // precomputed, do not recompute
    guard !pangramWords.isEmpty else { throw GeneratorError.emptyWordList }
    let wordSet = wordList.words

    for _ in 0..<maxAttempts {
        let pangram = pangramWords.randomElement()!
        let letters = Set(pangram)                    // exactly 7 unique chars
        for center in letters.shuffled() {
            let valid = wordSet.filter { isValidPuzzleWord($0, letters: letters, center: center) }
            if valid.count >= 20 {
                let pangrams = valid.filter { Set($0) == letters }
                return Puzzle(letters: letters, centerLetter: center,
                              validWords: valid, pangrams: pangrams)
            }
        }
    }
    throw GeneratorError.exhaustedRetries
}

private func isValidPuzzleWord(_ word: String, letters: Set<Character>, center: Character) -> Bool {
    guard word.count >= 4 else { return false }
    guard word.contains(center) else { return false }
    return Set(word).isSubset(of: letters)
}
```

Because construction starts from a 7-unique-letter word, the pangram list is guaranteed non-empty for any returned puzzle (D-04). Run Cmd+U with an iPhone 16 Simulator. All four tests must pass.
